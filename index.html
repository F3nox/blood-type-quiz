  // Generates an authentic glass lab slide SVG with Anti-A, Anti-B, and Anti-D wells
  function createBloodSlideSVG(antiA, antiB, antiD, sampleId) {
    function renderWell(cx, label, isClumped) {
      let bloodGraphic = '';
      if (isClumped) {
        // Agglutinated (clumped) red blood cells in clear plasma
        bloodGraphic = `
          <circle cx="${cx}" cy="68" r="26" fill="#3a161d" stroke="#64748b" stroke-width="1.5"/>
          <circle cx="${cx}" cy="68" r="24" fill="#e2e8f0" fill-opacity="0.22"/>
          <circle cx="${cx - 9}" cy="60" r="5.5" fill="#991b1b"/>
          <circle cx="${cx + 8}" cy="58" r="6" fill="#7f1d1d"/>
          <circle cx="${cx - 3}" cy="71" r="6.5" fill="#b91c1c"/>
          <circle cx="${cx + 10}" cy="74" r="5" fill="#991b1b"/>
          <circle cx="${cx - 11}" cy="76" r="4.5" fill="#7f1d1d"/>
          <circle cx="${cx + 1}" cy="57" r="4" fill="#dc2626"/>
          <circle cx="${cx + 3}" cy="81" r="4.5" fill="#7f1d1d"/>
        `;
      } else {
        // Smooth, uniform suspension (no reaction)
        bloodGraphic = `
          <circle cx="${cx}" cy="68" r="26" fill="#dc2626" stroke="#64748b" stroke-width="1.5"/>
          <ellipse cx="${cx - 7}" cy="59" rx="7" ry="4" transform="rotate(-25 ${cx - 7} 59)" fill="#f87171" fill-opacity="0.45"/>
        `;
      }

      return `
        <g>
          <text x="${cx}" y="30" text-anchor="middle" fill="#cbd5e1" font-size="12" font-weight="700" letter-spacing="0.5">${label}</text>
          ${bloodGraphic}
          <text x="${cx}" y="110" text-anchor="middle" fill="${isClumped ? '#fca5a5' : '#94a3b8'}" font-size="10" font-weight="600">
            ${isClumped ? 'CLUMPED (+)' : 'SMOOTH (-)'}
          </text>
        </g>
      `;
    }

    return `
      <div class="specimen-stage">
        <svg viewBox="0 0 380 128" width="100%" style="max-width: 440px; height: auto;">
          <!-- Glass Microscope Slide Base -->
          <rect x="6" y="8" width="368" height="112" rx="12" fill="rgba(148, 163, 184, 0.12)" stroke="rgba(148, 163, 184, 0.45)" stroke-width="2"/>
          <!-- Frosted Label Area -->
          <rect x="14" y="16" width="54" height="96" rx="6" fill="rgba(255, 255, 255, 0.08)" stroke="rgba(255,255,255,0.15)"/>
          <text x="41" y="55" text-anchor="middle" fill="#00d4ff" font-size="10" font-weight="700">SAMPLE</text>
          <text x="41" y="72" text-anchor="middle" fill="#f8fafc" font-size="12" font-weight="800">${sampleId}</text>

          ${renderWell(122, 'ANTI-A', antiA)}
          ${renderWell(214, 'ANTI-B', antiB)}
          ${renderWell(306, 'ANTI-D (Rh)', antiD)}
        </svg>
        <div class="slide-legend">
          <div class="legend-item"><span class="dot-clump"></span> Agglutination = Antigen Present (+)</div>
          <div class="legend-item"><span class="dot-smooth"></span> Smooth Mix = Antigen Absent (-)</div>
        </div>
      </div>
    `;
  }

  const qCounterEl = document.getElementById("q-counter");

  function loadQuestion() {
    answered = false;
    const q = questions[currentIdx];

    qCounterEl.textContent = `${currentIdx + 1} / ${questions.length}`;
    progressFillEl.style.width = `${((currentIdx + 1) / questions.length) * 100}%`;
    stepTagEl.textContent = q.tag;
    questionTextEl.textContent = q.question;

      visualContainerEl.innerHTML = createBloodSlideSVG(q.slide.a, q.slide.b, q.slide.d, q.slide.id);
    } else {
      visualContainerEl.innerHTML = `
        <div class="donor-banner">
          <div class="blood-bag-icon">${q.donorType}</div>
          <div class="donor-info">
            <h4>Confirmed Donor Unit</h4>
            <p>Select the exact compatible recipient group(s) for <strong>${q.donorType}</strong> red blood cells.</p>
          </div>
        </div>
      `;
    }

    // Reset feedback and next button
    feedbackBoxEl.className = "feedback-box";
    feedbackBoxEl.innerHTML = "";
    nextBtnEl.style.display = "none";
    nextBtnEl.textContent = currentIdx === questions.length - 1 ? "View Lab Report →" : "Next Lab Step →";

    // Render answer choices
    const letters = ["A", "B", "C", "D"];
    optionsGridEl.innerHTML = "";
    q.options.forEach((opt, idx) => {
      const btn = document.createElement("button");
      btn.className = "option-btn";
      btn.innerHTML = `<span class="option-key">${letters[idx]}</span><span>${opt}</span>`;
      btn.onclick = () => selectAnswer(idx, btn);
      optionsGridEl.appendChild(btn);
    });
  }

  function selectAnswer(selectedIdx, selectedBtn) {
    if (answered) return;
    answered = true;

    const q = questions[currentIdx];
    const allBtns = optionsGridEl.querySelectorAll(".option-btn");
    allBtns.forEach(b => b.disabled = true);

    if (selectedIdx === q.correct) {
      score++;
      scoreCounterEl.textContent = score;
      selectedBtn.classList.add("correct");
      feedbackBoxEl.className = "feedback-box show correct-fb";
      feedbackBoxEl.innerHTML = `<strong>✓ Verified Match:</strong> ${q.explanation}`;
    } else {
      selectedBtn.classList.add("wrong");
      allBtns[q.correct].classList.add("correct");
      feedbackBoxEl.className = "feedback-box show wrong-fb";
      feedbackBoxEl.innerHTML = `<strong>✕ Lab Alert:</strong> ${q.explanation}`;
    }

    nextBtnEl.style.display = "inline-block";
  }

  function nextQuestion() {
    currentIdx++;
    if (currentIdx < questions.length) {
      loadQuestion();
    } else {
      showResults();
    }
  }

    const titleEl = document.getElementById("result-title");
    const descEl = document.getElementById("result-desc");
